```
function parseBase64(str) {
  try {
    // return JSON.parse(Buffer.from(str, 'base64').toString('utf-8'));
    let txt = decodeURIComponent(escape(str));
    return JSON.parse(decodeURIComponent(escape(atob(txt))));
  } catch (e) {
    console.error('Failed to parse base64 project info:', e);
    return null;
  }
}

function encodeBase64(obj) {
  // 将字符串转换为 UTF-8 字节数组
  const encoder = new TextEncoder();
  const bytes = encoder.encode(JSON.stringify(obj));
  // 将字节数组转换为 btoa 可处理的 ASCII 字符串
  const asciiStr = String.fromCharCode(...bytes);
  // 进行 Base64 编码
  return btoa(asciiStr);  
}
```
